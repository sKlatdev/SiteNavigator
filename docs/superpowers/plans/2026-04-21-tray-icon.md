# Tray Icon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a system tray icon to `sitenavigator-win.exe` with Open, Restart, and Quit menu items.

**Architecture:** All changes live in the C# launcher (`scripts/PortableLauncher.cs`). The existing blocking `child.WaitForExit()` is replaced with a WinForms message pump (`Application.Run()`). A `NotifyIcon` with a `ContextMenuStrip` provides the tray UI. The build script gains two `/reference` flags to pull in `System.Windows.Forms` and `System.Drawing`.

**Tech Stack:** C# (.NET Framework 4.x), `System.Windows.Forms.NotifyIcon`, `System.Drawing.Graphics`, `@yao-pkg/pkg` portable build, PowerShell `csc.exe` compilation.

## Task 3: Extract `SpawnChild()` helper

**Files:**
- Modify: `scripts/PortableLauncher.cs`

The current `Main()` method inlines all the `ProcessStartInfo` setup. Extract it into a static helper so it can be called on startup and again on Restart.

- [ ] **Step 1: Add `SpawnChild()` as a new static method inside `Program`**

Add this method after the `Main()` method (before `ReadEmbeddedCorePayload`):

```csharp
private static Process SpawnChild(string coreExePath, string launcherDir)
{
    ProcessStartInfo startInfo = new ProcessStartInfo();
    startInfo.FileName = coreExePath;
    startInfo.WorkingDirectory = launcherDir;
    startInfo.UseShellExecute = false;
    startInfo.CreateNoWindow = true;
    startInfo.WindowStyle = ProcessWindowStyle.Hidden;

    string configuredDataDir = Environment.GetEnvironmentVariable("SITENAVIGATOR_DATA_DIR");
    startInfo.EnvironmentVariables["SITENAVIGATOR_DATA_DIR"] =
        string.IsNullOrWhiteSpace(configuredDataDir)
            ? Path.Combine(launcherDir, "data")
            : configuredDataDir;

    CopyEnvironmentVariable(startInfo, "PORT");
    CopyEnvironmentVariable(startInfo, "PORT_RETRY_COUNT");
    CopyEnvironmentVariable(startInfo, "ALLOWED_ORIGINS");
    CopyEnvironmentVariable(startInfo, "ENABLE_PATH_IMPORT");
    CopyEnvironmentVariable(startInfo, "ENABLE_INDEX_PATH_IO");
    CopyEnvironmentVariable(startInfo, "CONTENT_CACHE_MAX_AGE");
    CopyEnvironmentVariable(startInfo, "SLOW_ROUTE_MS");

    string openBrowser = Environment.GetEnvironmentVariable("SITENAVIGATOR_OPEN_BROWSER");
    startInfo.EnvironmentVariables["SITENAVIGATOR_OPEN_BROWSER"] =
        string.IsNullOrWhiteSpace(openBrowser) ? "true" : openBrowser;

    Process child = Process.Start(startInfo);
    if (child == null)
    {
        throw new InvalidOperationException("Failed to start packaged runtime.");
    }
    return child;
}
```

- [ ] **Step 2: Add `KillChild()` static method directly after `SpawnChild()`**

```csharp
private static void KillChild(Process child)
{
    try
    {
        if (!child.HasExited)
        {
            child.Kill();
        }
        child.WaitForExit(3000);
    }
    catch (InvalidOperationException)
    {
        // Process already exited — nothing to do.
    }
}
```

## Task 4: Add `CreateTrayIcon()` helper

**Files:**
- Modify: `scripts/PortableLauncher.cs`

- [ ] **Step 1: Add `CreateTrayIcon()` static method after `KillChild()`**

```csharp
private static NotifyIcon CreateTrayIcon(
    int port,
    string coreExePath,
    string launcherDir,
    SafeJobHandle jobHandle,
    ref Process currentChild)
{
    // Build a 16x16 teal icon with a white "S" glyph.
    Bitmap bmp = new Bitmap(16, 16);
    using (Graphics g = Graphics.FromImage(bmp))
    {
        g.Clear(Color.FromArgb(0x1a, 0x6b, 0x6b));
        using (Font font = new Font("Arial", 7f, FontStyle.Bold, GraphicsUnit.Point))
        using (SolidBrush brush = new SolidBrush(Color.White))
        {
            SizeF size = g.MeasureString("S", font);
            float x = (16f - size.Width) / 2f;
            float y = (16f - size.Height) / 2f;
            g.DrawString("S", font, brush, x, y);
        }
    }
    IntPtr hicon = bmp.GetHicon();
    Icon icon = Icon.FromHandle(hicon);

    ContextMenuStrip menu = new ContextMenuStrip();
    ToolStripMenuItem openItem = new ToolStripMenuItem("Open SiteNavigator");
    ToolStripMenuItem restartItem = new ToolStripMenuItem("Restart");
    ToolStripMenuItem quitItem = new ToolStripMenuItem("Quit");

    openItem.Click += (s, e) =>
    {
        try { Process.Start("http://localhost:" + port); }
        catch { }
    };

    restartItem.Click += (s, e) =>
    {
        _restarting = true;
        KillChild(currentChild);
        currentChild = SpawnChild(coreExePath, launcherDir);
        currentChild.EnableRaisingEvents = true;
        currentChild.Exited += (ps, pe) => OnChildExited();
        if (!AssignProcessToJobObject(jobHandle.DangerousGetHandle(), currentChild.Handle))
        {
            throw new InvalidOperationException("Failed to bind restarted runtime to process job.");
        }
        _restarting = false;
    };

    quitItem.Click += (s, e) =>
    {
        KillChild(currentChild);
        Application.Exit();
    };

    menu.Items.Add(openItem);
    menu.Items.Add(restartItem);
    menu.Items.Add(new ToolStripSeparator());
    menu.Items.Add(quitItem);

    NotifyIcon tray = new NotifyIcon();
    tray.Icon = icon;
    tray.Text = "SiteNavigator";
    tray.ContextMenuStrip = menu;
    tray.Visible = true;

    tray.DoubleClick += (s, e) =>
    {
        try { Process.Start("http://localhost:" + port); }
        catch { }
    };

    return tray;
}
```

- [ ] **Step 2: Add `OnChildExited()` static method after `CreateTrayIcon()`**

```csharp
private static void OnChildExited()
{
    if (_restarting) return;

    // Marshal back to UI thread for the balloon tip and exit.
    // Application.OpenForms may be empty; use a hidden form to invoke.
    Action cleanup = () =>
    {
        // Show balloon if a tray icon is still active.
        // The tray icon reference isn't stored globally, so we
        // find it via Application context — just exit cleanly.
        Application.Exit();
    };

    // Exited fires on a thread-pool thread; we need to post to the UI thread.
    try
    {
        // If there's a synchronization context (message pump is running), use it.
        System.Threading.SynchronizationContext ctx =
            System.Threading.SynchronizationContext.Current;
        if (ctx != null)
        {
            ctx.Post(_ => cleanup(), null);
        }
        else
        {
            cleanup();
        }
    }
    catch { }
}
```

> **Note:** `SynchronizationContext.Current` on the Exited thread-pool thread will be `null`. The correct approach is to capture the UI thread's context before `Application.Run()` and use it here. This is handled in Task 5.

Replace it entirely with:

```csharp
[STAThread]
private static int Main()
{
    Application.EnableVisualStyles();
    Application.SetCompatibleTextRenderingDefault(false);

    string launcherPath = Assembly.GetExecutingAssembly().Location;
    string launcherDir = Path.GetDirectoryName(launcherPath) ?? AppDomain.CurrentDomain.BaseDirectory;
    string runtimeRoot = Path.Combine(
        Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
        "SiteNavigator",
        "runtime"
    );

    Directory.CreateDirectory(runtimeRoot);

    byte[] payload = ReadEmbeddedCorePayload();
    string payloadHash = ComputeSha256Hex(payload);
    string runtimeDir = Path.Combine(runtimeRoot, payloadHash);
    string coreExePath = Path.Combine(runtimeDir, "sitenavigator-core.exe");
    Directory.CreateDirectory(runtimeDir);

    if (!File.Exists(coreExePath) || !FileContentMatches(coreExePath, payload))
    {
        File.WriteAllBytes(coreExePath, payload);
    }

    // Compute port using same rule as the Node server.
    string portEnv = Environment.GetEnvironmentVariable("PORT");
    _port = int.TryParse(portEnv, out int parsedPort) && parsedPort > 0 ? parsedPort : 8787;

    // Capture the UI thread's SynchronizationContext so OnChildExited can post back.
    System.Threading.SynchronizationContext uiContext =
        System.Threading.SynchronizationContext.Current
        ?? new System.Windows.Forms.WindowsFormsSynchronizationContext();
    System.Threading.SynchronizationContext.SetSynchronizationContext(uiContext);

    Process currentChild = SpawnChild(coreExePath, launcherDir);

    using (SafeJobHandle jobHandle = CreateKillOnCloseJob())
    {
        if (!AssignProcessToJobObject(jobHandle.DangerousGetHandle(), currentChild.Handle))
        {
            throw new InvalidOperationException("Failed to bind packaged runtime to process job.");
        }

        currentChild.EnableRaisingEvents = true;
        currentChild.Exited += (s, e) =>
        {
            if (_restarting) return;
            uiContext.Post(_ =>
            {
                Application.Exit();
            }, null);
        };

        using (NotifyIcon tray = CreateTrayIcon(_port, coreExePath, launcherDir, jobHandle, ref currentChild))
        {
            Application.Run();
        }
    }

    return 0;
}
```
